-- 13. REPORT: VOLUNTEER PARTICIPATION
SELECT
    v.Volunteer_ID,
    v.First_Name,
    v.Last_Name,
    d.Drive_ID,
    d.Drive_Name
FROM volunteer v
LEFT JOIN attends a
    ON v.Volunteer_ID = a.Volunteer_ID
LEFT JOIN drive d
    ON a.Drive_ID = d.Drive_ID
ORDER BY v.Volunteer_ID, d.Drive_ID;


-- 14. REPORT: TOTAL CERTIFIED HOURS
SELECT
    v.Volunteer_ID,
    v.First_Name,
    v.Last_Name,
    COALESCE(SUM(c.Hours), 0) AS Total_Hours
FROM volunteer v
LEFT JOIN certifies c
    ON v.Volunteer_ID = c.Volunteer_ID
GROUP BY
    v.Volunteer_ID,
    v.First_Name,
    v.Last_Name
ORDER BY v.Volunteer_ID;


-- 15. REPORT: DRIVES WITHOUT VOLUNTEERS
SELECT
    d.Drive_ID,
    d.Drive_Name,
    d.Cause
FROM drive d
LEFT JOIN attends a
    ON d.Drive_ID = a.Drive_ID
WHERE a.Volunteer_ID IS NULL
ORDER BY d.Drive_ID;
